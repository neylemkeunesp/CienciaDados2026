\section{Controle de Fluxo Idiomático}

\begin{frame}[fragile]{Laços Elegantes com \texttt{zip} e \texttt{enumerate}}
  Evite acessar elementos via índice manual \texttt{lista[i]}:
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Iteração Paralela com \texttt{zip}}
      \begin{lstlisting}
nomes = ['Ana', 'Beto']
notas = [9.5, 8.0]
for n, nota in zip(nomes, notas):
    print(f"{n}: {nota}")
      \end{lstlisting}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Contador com \texttt{enumerate}}
      \begin{lstlisting}
for idx, nome in enumerate(nomes, start=1):
    print(f"#{idx} - {nome}")
      \end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}

\section{Desenvolvimento de Funções Modulares}

\begin{frame}[fragile]{Anatomia de uma Função em Ciência de Dados}
  \begin{lstlisting}
def calcula_estatisticas(amostras: list[float], ddof: int = 1) -> dict[str, float]:
    """
    Calcula a média e o desvio padrão de uma amostra numérica.
    
    Parâmetros:
        amostras: Lista de observações numéricas.
        ddof: Graus de liberdade (padrão amostral = 1).
    """
    n = len(amostras)
    media = sum(amostras) / n
    var = sum((x - media)**2 for x in amostras) / (n - ddof)
    return {'media': media, 'desvio_padrao': var**0.5}
  \end{lstlisting}
\end{frame}

\section{Tratamento de Exceções}

\begin{frame}[fragile]{Robustez com \texttt{try / except}}
  Pipelines analíticos processam dados reais que frequentemente contêm valores corrompidos:
  \begin{lstlisting}
valores_brutos = ['10.5', '20.3', 'ausente', '45.1', 'N/A']
valores_limpos = []
for v in valores_brutos:
    try:
        valores_limpos.append(float(v))
    except ValueError:
        pass # Ignora ou registra em log de auditoria
  \end{lstlisting}
\end{frame}
